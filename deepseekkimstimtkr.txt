async def _fetch_json(url: str, params: dict, timeout: int = TIMEOUT) -> dict:
    try:
        async with aiohttp.ClientSession(headers=UA) as s:
            async with s.get(url, params=params, timeout=aiohttp.ClientTimeout(total=timeout)) as r:
                text = await r.text()
                if r.status != 200:
                    return {"_error": f"HTTP {r.status}", "_raw": text[:400]}
                try:
                    return json.loads(text)
                except json.JSONDecodeError:
                    return {"_raw": text[:2000]}
    except Exception as e:
        return {"_error": str(e)}

def esc(x) -> str:
    return html.escape(str(x), quote=False)

def flatten(obj, prefix: str = "") -> list:
    out = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            key = f"{prefix}{k}"
            if isinstance(v, (dict, list)):
                out += flatten(v, f"{key}.")
            else:
                out.append((key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            if isinstance(v, (dict, list)):
                out += flatten(v, f"{prefix}{i}.")
            else:
                out.append((f"{prefix}{i}", v))
    return out

def kv_block(data, max_fields: int = 50, max_val: int = 100) -> str:
    rows = flatten(data)
    lines = []
    shown = 0
    for k, v in rows:
        if v is None or v == "" or v == [] or v == {}:
            continue
        s = str(v)
        if len(s) > max_val:
            s = s[:max_val] + "…"
        lines.append(f"▪ <b>{esc(k)}</b>: <code>{esc(s)}</code>")
        shown += 1
        if shown >= max_fields:
            lines.append(f"<i>… {len(rows)} fields total</i>")
            break
    return "\n".join(lines)

async def safe_edit(msg, text: str):
    if len(text) > MAX_LEN:
        text = text[:MAX_LEN] + "\n<i>… truncated</i>"
    try:
        await msg.edit_text(text, parse_mode=ParseMode.HTML, disable_web_page_preview=True)
    except Exception:
        plain = re.sub(r"<[^>]+>", "", text)
        try:
            await msg.edit_text(plain, disable_web_page_preview=True)
        except Exception:
            pass

async def render_menu(message, page: int, edit: bool = True):
    pg = MENU_PAGES[page]
    kb = []
    for row in pg["buttons"]:
        kb_row = []
        for label, cb, emoji in row:
            kb_row.append(InlineKeyboardButton(f"{emoji} {label}", callback_data=cb))
        kb.append(kb_row)

    nav = []
    if page > 0:
        nav.append(InlineKeyboardButton("« PREV", callback_data="page:prev"))
    else:
        nav.append(InlineKeyboardButton("🏠 HOME", callback_data="page:home"))
    
    if page < len(MENU_PAGES) - 1:
        nav.append(InlineKeyboardButton("NEXT »", callback_data="page:next"))
    else:
        nav.append(InlineKeyboardButton("🔄 RELOAD", callback_data="page:home"))
    kb.append(nav)

    text = (
        f"{pg['title']}\n"
        f"{DIV}\n\n"
        f"{em(EMOJI_CROWN, '👑')} <i>Bot made by: @DARK_OSINT5</i>"
    )

    if edit:
        try:
            await message.edit_text(text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.HTML)
        except Exception:
            pass
    else:
        await message.reply_text(text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.HTML)

        f"{em(EMOJI_GEAR, '⚙️')} Searching <code>{esc(query)}</code> ...",

async def run_gbs(msg, cmd, query):
    data = await gbs(cmd, query)
    out = f"🔍 <b>{cmd.upper()}</b> — <code>{esc(query)}</code>\n{DIV}\n\n"
    if "_error" in data:
        out += f"{em(EMOJI_CROSS, '❌')} <b>Error:</b> <code>{esc(data['_error'])}</code>"
    elif "_raw" in data:
        out += f"{em(EMOJI_WARNING, '⚠️')} <b>Raw output:</b>\n<code>{esc(data['_raw'][:1000])}</code>"
    else:
        out += kv_block(data, max_fields=40) or "❌ empty response"
    await safe_edit(msg, out)

async def run_combined_vehicle(msg, vehicle):
    vehicle = vehicle.upper()
    leak = await leak_vehicle(vehicle)
    rc = await gbs("rc", vehicle)
    rc2 = await gbs("rc2", vehicle)
    
    out = f"🚗 <b>VEHICLE INFO</b> — <code>{esc(vehicle)}</code>\n{DIV}\n\n"
    
    out += f"{em(EMOJI_FIRE, '🔥')} <b>LEAKINFO V6</b>\n"
    if "_error" in leak:
        out += f"❌ error: {esc(leak['_error'])}\n"
    elif not leak.get("success"):
        out += f"❌ no data\n"
    else:
        data = leak.get("data", {})
        out += kv_block(data, max_fields=15)
        rl = (leak.get("rate_limits") or [{}])[0]
        if rl.get("remaining") is not None:
            out += f"\n<i>rate: {rl.get('remaining')}/{rl.get('limit')}</i>"
    out += f"\n{DIV}\n"
    
    out += f"{em(EMOJI_SHIELD, '🛡')} <b>GBS RC</b>\n"
    if "_error" in rc:
        out += f"❌ error: {esc(rc['_error'])}\n"
    else:
        out += kv_block(rc, max_fields=15) or "❌ empty\n"
    out += f"\n{DIV}\n"
    
    out += f"{em(EMOJI_SHIELD, '🛡')} <b>GBS RC v2</b>\n"
    if "_error" in rc2:
        out += f"❌ error: {esc(rc2['_error'])}\n"
    else:
        out += kv_block(rc2, max_fields=15) or "❌ empty\n"
        
    await safe_edit(msg, out)

async def run_all(msg, query):
    qs = query.strip()
    digits = qs.replace("+","").replace("-","").replace(" ","")
    out = f"{em(EMOJI_BELL, '🌐')} <b>SMART SEARCH</b> — <code>{esc(query)}</code>\n{DIV}\n\n"
    
    plan = []
    if digits.isdigit():
        plan = [("phoneinfo", "Phone Info"), ("number", "Number Lookup"), ("numtoupi", "UPI")]
        if len(digits) == 12:
            plan += [("ration", "Ration"), ("family", "Family")]
    elif len(qs) == 10 and qs[:5].isalpha() and qs[5:].isdigit():
        plan = [("pan", "PAN"), ("pan2", "PAN v2")]
    elif qs.count(".") == 3 and all(p.isdigit() for p in qs.split(".")):
        plan = [("ip", "IP Info")]
    else:
        v = qs.upper()
        plan = [("rc", "Vehicle RC"), ("rc2", "RC v2"), ("vehiclenum", "Veh Number"), ("v2n", "V2N"), ("smcv1", "SMC"), ("vehiclechallan", "Challan")]
        leak = await leak_vehicle(v)
        out += f"{em(EMOJI_FIRE, '🔥')} <b>LEAKINFO V6</b>\n"
        if not leak.get("_error") and leak.get("success"):
            out += kv_block(leak.get("data", {}), max_fields=10) + "\n\n"
        else:
            out += "❌ no data\n\n"

    for cmd, label in plan:
        out += f"{em(EMOJI_SHIELD, '🛡')} <b>{label}</b>\n"
        data = await gbs(cmd, qs.upper() if cmd in ["rc", "rc2", "vehiclenum", "v2n", "smcv1", "vehiclechallan"] else qs)
        if "_error" in data:
            out += f"❌ {esc(data['_error'])}\n\n"
        else:
            block = kv_block(data, max_fields=8, max_val=50)
            out += (block or "❌ empty") + "\n\n"
            
    await safe_edit(msg, out)
